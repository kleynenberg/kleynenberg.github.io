def compute_hashes(file_path: Path) -> tuple[str, str]:
    md5_h = hashlib.md5()
    sha256_h = hashlib.sha256()
    with file_path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(65536), b""):
            md5_h.update(chunk)
            sha256_h.update(chunk)
    return md5_h.hexdigest(), sha256_h.hexdigest()


def run_cmd(cmd: list[str]) -> subprocess.CompletedProcess:
    return subprocess.run(cmd, capture_output=True, text=True, check=False)


def get_pdf_page_count(pdf_path: Path) -> int:
    proc = run_cmd(["pdfinfo", str(pdf_path)])
    if proc.returncode != 0:
        raise RuntimeError(f"pdfinfo fout op {pdf_path.name}: {proc.stderr.strip()}")
    for line in proc.stdout.splitlines():
        if line.startswith("Pages:"):
            return int(line.split(":", 1)[1].strip())
    return 0


def extract_pdf_pages(pdf_path: Path) -> list[str]:
    proc = run_cmd(["pdftotext", "-enc", "UTF-8", str(pdf_path), "-"])
    if proc.returncode != 0:
        raise RuntimeError(f"pdftotext fout op {pdf_path.name}: {proc.stderr.strip()}")
    normalized = proc.stdout.replace("\r\n", "\n").replace("\r", "\n")
    pages = normalized.split("\x0c")
    if pages and pages[-1].strip() == "":
        pages.pop()
    return pages


def non_ws_len(text: str) -> int:
    return len(re.sub(r"\s+", "", text))


def process_pdf(src_pdf: Path, md5_hex: str) -> tuple[str, int, list[str], str]:
    page_count = get_pdf_page_count(src_pdf)
    initial_pages = extract_pdf_pages(src_pdf)
    total_chars = sum(non_ws_len(p) for p in initial_pages)
    avg_chars = total_chars / max(page_count, 1)
    sparse_pages = sum(1 for p in initial_pages if non_ws_len(p) < 40)

    needs_ocr = (page_count > 0) and (avg_chars < 80 or sparse_pages > (page_count / 2))
    if not needs_ocr:
        return "DIRECT_TEXT_LAYER", page_count, initial_pages, "Geen OCR nodig (volledige tekstlaag aanwezig)"

    ocr_pdf_path = OCR_PDF_DIR / f"{md5_hex}.pdf"
    ocr_cmd = [
        "ocrmypdf",
        "-l",
        "eng+nld",
        "--deskew",
        "--clean",
        "--force-ocr",
        "--optimize",
        "1",
        str(src_pdf),
        str(ocr_pdf_path),
    ]
    proc = run_cmd(ocr_cmd)
    note = "OCRmyPDF 16.13.0 (-l eng+nld --deskew --clean --force-ocr)"
    if proc.returncode != 0:
        fallback_cmd = [
            "ocrmypdf",
            "-l",
            "eng+nld",
            "--deskew",
            "--force-ocr",
            str(src_pdf),
            str(ocr_pdf_path),
        ]
        proc = run_cmd(fallback_cmd)
        note = "OCRmyPDF 16.13.0 (-l eng+nld --deskew --force-ocr, zonder --clean)"
        if proc.returncode != 0:
            raise RuntimeError(f"OCRmyPDF fout op {src_pdf.name}: {proc.stderr.strip()}")

    ocr_pages = extract_pdf_pages(ocr_pdf_path)
    return "OCR_APPLIED", page_count, ocr_pages, note


def read_normalized_text(file_path: Path) -> str:
    raw = file_path.read_bytes()
    try:
        text = raw.decode("utf-8-sig")
    except UnicodeDecodeError:
        text = raw.decode("latin-1")
    return text.replace("\r\n", "\n").replace("\r", "\n")


def parse_xml_patent_or_generic(xml_path: Path) -> tuple[str, dict]:
    parser = etree.XMLParser(recover=True, remove_blank_text=True)
    tree = etree.parse(str(xml_path), parser)
    root = tree.getroot()
    root_tag = etree.QName(root).localname if isinstance(root.tag, str) else str(root.tag)

    lines = [f"XML Root: <{root_tag}>"]
    meta = {"xml_root": root_tag}

    # Verzamel alle betekenisvolle tekstblokken met lokale tagnamen voor context
    seen_blocks = set()
    for elem in root.iter():
        if not isinstance(elem.tag, str):
            continue
        local = etree.QName(elem).localname
        if local in {
            "invention-title",
            "doc-number",
            "kind",
            "country",
            "date",
            "name",
            "orgname",
            "last-name",
            "first-name",
            "classification-ipcr",
            "main-classification",
            "abstract",
            "p",
            "claim-text",
            "heading",
        }:
            txt = " ".join(" ".join(elem.itertext()).split())
            if txt and txt not in seen_blocks:
                seen_blocks.add(txt)
                lines.append(f"[{local}] {txt}")

    if len(lines) <= 2:
        full_txt = "\n".join(
            " ".join(line.split())
            for line in "".join(root.itertext()).splitlines()
            if line.strip()
        )
        lines.append(full_txt)

    return "\n".join(lines), meta


def extract_years(text: str) -> list[int]:
    years = {int(m) for m in YEAR_RE.findall(text) if 1980 <= int(m) <= 2026}
    return sorted(years)


def extract_dois(text: str) -> list[str]:
    cleaned = set()
    for raw in DOI_RE.findall(text):
        doi = raw.rstrip(".,;:)]}>\"'")
        if len(doi) > 8:
            cleaned.add(doi)
    return sorted(cleaned)


def extract_patents(text: str) -> list[str]:
    found = set()
    for raw in PATENT_RE.findall(text):
        norm = re.sub(r"\s+", " ", raw.strip())
        if len(norm) >= 6:
            found.add(norm)
    return sorted(found)


def extract_eu_grants(text: str) -> list[str]:
    found = {re.sub(r"\s+", " ", m.strip()).upper() for m in EU_GRANT_RE.findall(text)}
    return sorted(found)


def detect_title(text: str, fallback_name: str) -> str:
    for line in text.splitlines():
        s = line.strip().lstrip("#").strip()
        if not s or s.startswith("---"):
            continue
        if len(s) >= 4:
            return s[:160]
    return fallback_name


def extract_snippets(pages_or_sections: list[tuple[str, str]], doc_md5: str, primary_name: str) -> list[dict]:
    snippets = []
    seen_quotes = set()
    for loc_label, block_text in pages_or_sections:
        # Splits op alinea's of losse regels zodat zowel tabellen/lists als lopende tekst behouden blijven
        chunks = re.split(r"\n\s*\n", block_text)
        for chunk in chunks:
            sub_units = [chunk] if len(chunk) <= 650 else chunk.splitlines()
            for unit in sub_units:
                clean_u = " ".join(unit.split()).strip()
                if len(clean_u) < 18:
                    continue
                yrs = extract_years(clean_u)
                dois = extract_dois(clean_u)
                pats = extract_patents(clean_u)
                grants = extract_eu_grants(clean_u)
                if not (yrs or dois or pats or grants):
                    continue
                if len(clean_u) > 520:
                    clean_u = clean_u[:517] + "..."
                dedup_key = (loc_label, clean_u)
                if dedup_key in seen_quotes:
                    continue
                seen_quotes.add(dedup_key)
                snippets.append(
                    {
                        "doc_md5": doc_md5,
                        "source_file": primary_name,
                        "location": loc_label,
                        "years": yrs,
                        "dois": dois,
                        "patents": pats,
                        "eu_or_grant_signals": grants,
                        "corpus_attested": True,
                        "quote": clean_u,
                    }
                )
    return snippets
